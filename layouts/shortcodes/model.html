{{- /* model shortcode -- embeds an interactive glTF/GLB viewer via . Params: src (required) glTF/GLB file, resolved as a page resource or from assets/ alt accessible description caption figcaption text (markdown) poster image shown before the model loads (page resource) height CSS height of the viewer, default 460px autorotate "false" to disable the idle turntable */ -}} {{- $srcRaw := .Get "src" -}} {{- if not $srcRaw -}} {{- errorf "model shortcode: src is required: %s" .Position -}} {{- end -}} {{- $src := "" -}} {{- with $.Page.Resources.GetMatch $srcRaw -}} {{- $src = .RelPermalink -}} {{- else -}} {{- with resources.Get $srcRaw -}} {{- $src = .RelPermalink -}} {{- else -}} {{- if fileExists (path.Join "static" $srcRaw) -}} {{- $src = relURL $srcRaw -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $src -}} {{- errorf "model shortcode: unable to resolve src %q: %s" $srcRaw .Position -}} {{- end -}} {{- $poster := "" -}} {{- with .Get "poster" -}} {{- with $.Page.Resources.GetMatch . -}} {{- $poster = .RelPermalink -}} {{- else -}} {{- warnf "model shortcode: unable to resolve poster %q: %s" . $.Position -}} {{- end -}} {{- end -}} {{- $alt := .Get "alt" | default "Interactive 3D model" -}} {{- $height := .Get "height" | default "460px" -}} {{- $autorotate := true -}} {{- with .Get "autorotate" -}} {{- $autorotate = not (in (slice "false" "0" "no" "off") (lower .)) -}} {{- end -}} {{- $caption := .Get "caption" -}}
